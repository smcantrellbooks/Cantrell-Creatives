<script>
(function() {
  var primary  = 'https://cdn.jsdelivr.net/npm/@supabase/supabase-js@2.45.4/dist/umd/supabase.js';
  var fallback = 'https://unpkg.com/@supabase/supabase-js@2.45.4/dist/umd/supabase.js';

  // ── CONFIG — replace these ──
  var SUPABASE_URL      = 'https://recyyhvdvuupxhbxmzal.supabase.co';
  var SUPABASE_ANON_KEY = 'eyJhbGciOiJIUzI1NiIsInR5cCI6IkpXVCJ9.eyJpc3MiOiJzdXBhYmFzZSIsInJlZiI6InJlY3l5aHZkdnV1cHhoYnhtemFsIiwicm9sZSI6ImFub24iLCJpYXQiOjE3NzQxMjc0MDAsImV4cCI6MjA4OTcwMzQwMH0.kO4niR2v4hgPFi2iQ8R8I1Id_COcRH_z4SsXc6W93AM';

  var STRIPE_LINKS = {
    visual:      'https://buy.stripe.com/VISUAL_LINK',
    publication: 'https://buy.stripe.com/PUBLICATION_LINK'
  };
  var TIER_DESTINATIONS = {
    free:        'https://creatives.smcantrellbooks.com/studio',
    visual:      'https://creatives.smcantrellbooks.com/studio',
    publication: 'https://creatives.smcantrellbooks.com/publisher-workspace',
    reader:      'https://smcantrellbooks.com'
  };

  function showState(id) {
    ['confirming','success','stripe','error'].forEach(function(s) {
      var el = document.getElementById('state-' + s);
      if (el) el.className = 'state' + (s === id ? ' show' : '');
    });
  }

  function animateProg(id, ms, cb) {
    var el = document.getElementById('prog-' + id);
    if (!el) return cb && cb();
    var w = 0;
    var step = 100 / (ms / 50);
    var t = setInterval(function() {
      w = Math.min(w + step, 100);
      el.style.width = w + '%';
      if (w >= 100) { clearInterval(t); if (cb) cb(); }
    }, 50);
  }

  function showError(msg) {
    showState('error');
    var el = document.getElementById('err-msg');
    if (el && msg) el.textContent = msg;
  }

  function loadScript(src, ok, fail) {
    var s = document.createElement('script');
    s.src = src; s.onload = ok; s.onerror = fail;
    document.head.appendChild(s);
  }

  function init() {
    if (typeof window.supabase === 'undefined') {
      return showError('Authentication library failed to load. Please refresh the page.');
    }

    var sb = window.supabase.createClient(SUPABASE_URL, SUPABASE_ANON_KEY);

    // Start confirming animation
    animateProg('confirming', 1800);

    // Give Supabase time to process the URL hash token
    setTimeout(function() {
      sb.auth.getSession().then(function(res) {
        var session = res.data && res.data.session;

        if (session) {
          afterConfirm(sb, session.user);
          return;
        }

        // Try manually reading tokens from URL hash (email link flow)
        var hash   = window.location.hash.replace('#', '');
        var params = new URLSearchParams(hash);
        var access  = params.get('access_token');
        var refresh = params.get('refresh_token');
        var type    = params.get('type');

        if (access) {
          sb.auth.setSession({ access_token: access, refresh_token: refresh || '' }).then(function(r) {
            if (r.error || !r.data || !r.data.session) {
              return showError(r.error ? r.error.message : 'Confirmation failed. Please try signing up again.');
            }
            afterConfirm(sb, r.data.session.user);
          });
        } else {
          // Check URL search params too (some OAuth flows use query string)
          var searchParams = new URLSearchParams(window.location.search);
          var code = searchParams.get('code');
          if (code) {
            sb.auth.exchangeCodeForSession(code).then(function(r) {
              if (r.error || !r.data || !r.data.session) {
                return showError(r.error ? r.error.message : 'Confirmation failed.');
              }
              afterConfirm(sb, r.data.session.user);
            });
          } else {
            // Last attempt — wait a bit longer for Supabase to process
            setTimeout(function(){
              sb.auth.getSession().then(function(res2){
                if(res2.data && res2.data.session){
                  afterConfirm(sb, res2.data.session.user);
                } else {
                  showError('No confirmation token found. Please use the link from your email or try signing in directly.');
                }
              });
            }, 1500);
          }
        }
      }).catch(function(e) {
        showError('Something went wrong: ' + (e.message || 'unknown error'));
      });
    }, 800);
  }

  function afterConfirm(sb, user) {
    sb.from('members').select('tier, first_name').eq('id', user.id).single().then(function(memberRes) {
      var tier      = (memberRes.data && memberRes.data.tier)       ? memberRes.data.tier       : (localStorage.getItem('cc_pending_tier') || 'free');
      var firstName = (memberRes.data && memberRes.data.first_name) ? memberRes.data.first_name : '';

      localStorage.setItem('cc_user_tier',  tier);
      localStorage.setItem('cc_user_name',  firstName);
      localStorage.setItem('cc_user_email', user.email);
      localStorage.setItem('cc_user_id',    user.id);

      var pendingStripeUrl = localStorage.getItem('cc_pending_stripe_url');
      var needsStripe = (tier === 'visual' || tier === 'publication' || tier === 'reader') && pendingStripeUrl;

      if (needsStripe) {
        showState('stripe');
        var badge = document.getElementById('stripe-tier-name');
        if (badge) badge.textContent = tier.charAt(0).toUpperCase() + tier.slice(1) + ' Membership';

        animateProg('stripe', 2500, function() {
          localStorage.removeItem('cc_pending_tier');
          localStorage.removeItem('cc_pending_stripe_url');
          localStorage.removeItem('cc_pending_billing');
          // Add prefilled email and client reference to Stripe URL
          var url = pendingStripeUrl;
          var sep = url.indexOf('?') > -1 ? '&' : '?';
          url += sep + 'prefilled_email=' + encodeURIComponent(user.email) + '&client_reference_id=' + user.id;
          window.location.href = url;
        });
      } else {
        showState('success');
        animateProg('success', 2500, function() {
          window.location.href = TIER_DESTINATIONS[tier] || TIER_DESTINATIONS.free;
        });
      }
    }).catch(function() {
      // Member record not found — still redirect to studio as free
      showState('success');
      animateProg('success', 2500, function() {
        window.location.href = TIER_DESTINATIONS.free;
      });
    });
  }

  loadScript(primary, init, function() {
    loadScript(fallback, init, function() {
      showError('Authentication library could not be loaded. Please refresh the page.');
    });
  });
})();
</script>
