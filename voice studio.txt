// ── FILE UPLOAD ──
function handleDrop(e){
  e.preventDefault();
  document.getElementById('uploadZone').classList.remove('drag-over');
  var file = e.dataTransfer.files[0];
  if(file) processFile(file);
}
function handleFileUpload(e){
  var file = e.target.files[0];
  if(file) processFile(file);
  e.target.value='';
}
function processFile(file){
  var ext = file.name.split('.').pop().toLowerCase();
  showStatus('uploadStatus','info','Reading '+file.name+'…');
  if(!document.getElementById('bookTitle').value)
    document.getElementById('bookTitle').value = file.name.replace(/\.[^.]+$/,'');

  if(ext==='txt'){
    var r = new FileReader();
    r.onload = function(ev){
      chapters = splitIntoChapters(ev.target.result);
      renderChapters();
      showStatus('uploadStatus','ok','&#10003; '+file.name+' loaded — '+chapters.length+' chapters detected');
    };
    r.readAsText(file);
  } else if(ext==='epub'||ext==='docx'){
    if(typeof JSZip==='undefined'){ showStatus('uploadStatus','err','JSZip not loaded. Please refresh.'); return; }
    var r2 = new FileReader();
    r2.onload = function(ev){
      JSZip.loadAsync(ev.target.result).then(function(zip){
        var textPromises = [];
        if(ext==='epub'){
          var files = Object.keys(zip.files).filter(function(f){ return (f.endsWith('.html')||f.endsWith('.xhtml'))&&!f.includes('toc'); }).sort();
          files.forEach(function(fn){ textPromises.push(zip.files[fn].async('text')); });
        } else {
          var doc = zip.files['word/document.xml'];
          if(doc) textPromises.push(doc.async('text'));
        }
        return Promise.all(textPromises);
      }).then(function(arr){
        var full = arr.map(function(h){ return h.replace(/<[^>]+>/g,' ').replace(/\s+/g,' ').trim(); }).join('\n\n');
        chapters = splitIntoChapters(full);
        renderChapters();
        showStatus('uploadStatus','ok','&#10003; '+file.name+' loaded — '+chapters.length+' chapters');
      }).catch(function(e){ showStatus('uploadStatus','err','Error reading file: '+e.message); });
    };
    r2.readAsArrayBuffer(file);
  }
}

function splitIntoChapters(text){
  // Try to detect chapter headings
  var chapterRegex = /(?:^|\n)(chapter\s+\d+[^\n]*|part\s+\d+[^\n]*|prologue|epilogue)/gi;
  var matches = [], match;
  while((match = chapterRegex.exec(text)) !== null) matches.push({title: match[1].trim(), index: match.index});

  if(matches.length > 1){
    return matches.map(function(m, i){
      var end = i+1 < matches.length ? matches[i+1].index : text.length;
      return { title: m.title, text: text.substring(m.index, end).trim() };
    });
  }
  // No chapters detected — split by word count (~2000 words per chapter)
  var words = text.split(/\s+/);
  var result = [], chunkSize = 2000;
  for(var i=0; i<words.length; i+=chunkSize){
    var chunk = words.slice(i, i+chunkSize).join(' ');
    result.push({ title: 'Chapter '+(Math.floor(i/chunkSize)+1), text: chunk });
  }
  return result.length ? result : [{title:'Chapter 1', text:text}];
}

function renderChapters(){

// ── GENERATE AUDIOBOOK ──
async function generateAudiobook(){
  if(!chapters.length){ showStatus('genStatus','err','Please upload a manuscript first.'); return; }
  var title  = document.getElementById('bookTitle').value.trim()||'Untitled';
  var author = document.getElementById('authorName').value.trim()||'';
  var provider = document.getElementById('ttsProvider').value;
  var speed  = document.querySelector('input[type=range]:nth-of-type(3)').value / 100;

  var btn = document.getElementById('generateBtn');
  btn.disabled = true; btn.textContent = '&#127911; Generating…';

  var progress = document.getElementById('genProgress');
  var fill = document.getElementById('genProgressFill');
  progress.style.display = 'block'; fill.style.width = '5%';
  showStatus('genStatus','info','Sending manuscript to audiobook engine…');

  try {
    var fullText = chapters.map(function(ch){ return ch.title+'\n\n'+ch.text; }).join('\n\n\n');
    fill.style.width = '20%';

    var endpoint, body, headers = {'Content-Type':'application/json'};

    if(provider === 'render'){
      // Render audiobook-creator endpoint
      endpoint = RENDER_URL + '/generate';
      body = JSON.stringify({
        title:    title,
        author:   author,
        text:     fullText,
        voice:    selectedVoice,
        speed:    speed,
        chapters: chapters.map(function(ch){ return {title:ch.title, text:ch.text}; })
      });
    } else {
      // Cloudflare Worker TTS
      endpoint = WORKER_BASE + '/tts';
      body = JSON.stringify({
        text:     fullText.substring(0, 5000),
        voice:    selectedVoice,
        provider: provider
      });
    }

    fill.style.width = '50%';
    showStatus('genStatus','info','Processing with '+provider+'… this may take a moment.');

    var resp = await fetch(endpoint, { method:'POST', headers:headers, body:body });
    fill.style.width = '85%';

    if(!resp.ok) throw new Error('Server returned '+resp.status+': '+resp.statusText);

    var contentType = resp.headers.get('content-type')||'';
    if(contentType.includes('audio') || contentType.includes('octet')){
      var blob = await resp.blob();
      generatedAudioUrl = URL.createObjectURL(blob);
    } else {
      var data = await resp.json();
      generatedAudioUrl = data.audio_url || data.url || data.file_url || null;
      if(!generatedAudioUrl && data.message) throw new Error(data.message);
    }

    fill.style.width = '100%';

    if(generatedAudioUrl){
      document.getElementById('generatedAudio').src = generatedAudioUrl;
      document.getElementById('audioResultBox').style.display = 'block';
      showStatus('genStatus','ok','&#10003; Audiobook generated successfully!');
      // Also save to localStorage for collections
      localStorage.setItem('collections_incoming', JSON.stringify({ title:title, author:author, audio_url:generatedAudioUrl }));
    } else {
      showStatus('genStatus','err','Generated but no audio URL returned. Check provider settings.');
    }

  } catch(err){
    showStatus('genStatus','err','Error: '+err.message);
    fill.style.width = '0%';
  }

  btn.disabled = false; btn.innerHTML = '&#127911; Generate Audiobook';
}

function downloadAudio(){
  if(!generatedAudioUrl) return;
  var a = document.createElement('a');
  a.href = generatedAudioUrl;
  a.download = (document.getElementById('bookTitle').value||'audiobook')+'.mp3';
  a.click();
}

async function generateTTS(){
  var text = document.getElementById('ttsText').value.trim();
  if(!text){ showStatus('ttsStatus','err','Please enter some text.'); return; }
  var voice    = document.getElementById('ttsVoice').value;
  var provider = document.getElementById('ttsSingleProvider').value;
  showStatus('ttsStatus','info','Generating speech…');

  try{
    var endpoint = provider==='render' ? RENDER_URL+'/tts' : WORKER_BASE+'/tts';
    var resp = await fetch(endpoint, {
      method:'POST', headers:{'Content-Type':'application/json'},
      body: JSON.stringify({ text:text, voice:voice, provider:provider })
    });
    if(!resp.ok) throw new Error(resp.status+' '+resp.statusText);
    var contentType = resp.headers.get('content-type')||'';
    var audioUrl;
    if(contentType.includes('audio')||contentType.includes('octet')){
      var blob = await resp.blob();
      audioUrl = URL.createObjectURL(blob);
    } else {
      var data = await resp.json();
      audioUrl = data.audio_url||data.url||null;
    }
    if(audioUrl){
      document.getElementById('ttsAudioBox').innerHTML = '<audio controls style="width:100%;" src="'+audioUrl+'"></audio><div style="margin-top:8px;"><a href="'+audioUrl+'" download="speech.mp3" class="btn-sec" style="display:inline-flex;">&#11015; Download</a></div>';
      showStatus('ttsStatus','ok','&#10003; Speech generated!');
    }
  }catch(e){ showStatus('ttsStatus','err','Error: '+e.message); }
}
