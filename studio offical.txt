// Toast notifications

  const toast = document.getElementById('T');

// Image generation - Pollinations.ai (free, no key needed)
function generateImage() {
  var prompt = document.getElementById('promptInput').value.trim();
  if (!prompt) { showToast('Please enter a description', 'err'); return; }

  var lc  = document.getElementById('LC');
  var es  = document.getElementById('ES');
  var img = document.getElementById('cImg');

  es.classList.add('hide');
  lc.style.cssText = 'display:flex;position:absolute;inset:0;background:rgba(0,0,0,.6);flex-direction:column;align-items:center;justify-content:center;gap:14px;z-index:20;border-radius:10px;';

  var styleVal   = (document.getElementById('imageStyle') || {}).value || 'cinematic realism';
  var fullPrompt = prompt + ', ' + styleVal + ', high quality, detailed';
  var seed       = Math.floor(Math.random() * 999999);
  var imgUrl     = 'https://image.pollinations.ai/prompt/'
    + encodeURIComponent(fullPrompt)
    + '?width=1280&height=720&seed=' + seed + '&nologo=true&enhance=true';

  // Use a hidden loader — no crossOrigin
  var loader = new Image();
  loader.onload = function() {
    lc.style.display = 'none';
    img.removeAttribute('crossorigin');
    img.src = imgUrl;
    img.classList.add('vis');
    es.classList.add('hide');
    img.style.cssText = 'display:block;position:absolute;inset:0;width:100%;height:100%;object-fit:contain;z-index:10;border-radius:10px;';
    addAsset({ type: 'image', src: imgUrl, name: prompt.substring(0, 30) });
    showToast('Image generated!', 'ok');
  };
  loader.onerror = function() {
    // Retry with different seed
    var retryUrl = 'https://image.pollinations.ai/prompt/'
      + encodeURIComponent(fullPrompt)
      + '?width=1280&height=720&seed=' + (seed + 7) + '&nologo=true';
    var loader2 = new Image();
    loader2.onload = function() {
      lc.style.display = 'none';
      img.removeAttribute('crossorigin');
      img.src = retryUrl;
      img.classList.add('vis');
      es.classList.add('hide');
      img.style.cssText = 'display:block;position:absolute;inset:0;width:100%;height:100%;object-fit:contain;z-index:10;border-radius:10px;';
      addAsset({ type: 'image', src: retryUrl, name: prompt.substring(0, 30) });
      showToast('Image generated!', 'ok');
    };
    loader2.onerror = function() {
      lc.style.display = 'none';
      es.classList.remove('hide');
      showToast('Generation failed — check connection', 'err');
    };
    loader2.src = retryUrl;
  };
  loader.src = imgUrl;
}

function handleImageUpload(event) {
  const file = event.target.files[0];
  if (!file) return;

  const reader = new FileReader();
  reader.onload = function(e) {
    // Switch to image tab first
    switchTab('I');

    // Then set the image after the panel is visible
    setTimeout(() => {
      const cb  = document.getElementById('CB');
      const es  = document.getElementById('ES');
      const lc  = document.getElementById('LC');

      // Remove any old generated result
      const old = document.getElementById('genResult');
      if (old) old.remove();

      // Create image element inside canvas box
      const img = document.getElementById('cImg');
      img.src            = e.target.result;
      img.style.display  = 'block';
      img.style.position = 'absolute';

// Video generation - Pollinations.ai cinematic frame (free, no key needed)
function generateVideo() {
  var prompt = document.getElementById('videoPromptInput').value.trim();
  if (!prompt) { showToast('Please enter a video description', 'err'); return; }

  var vlc = document.getElementById('VLC');
  var ves = document.getElementById('VES');
  var vb  = document.getElementById('VB');

  ves.style.display = 'none';
  vlc.style.cssText = 'display:flex;position:absolute;inset:0;background:rgba(0,0,0,.6);flex-direction:column;align-items:center;justify-content:center;gap:14px;z-index:20;border-radius:10px;';

  var old = document.getElementById('vidResult');
  if (old) old.remove();

  var styleVal   = document.getElementById('videoStyle') ? document.getElementById('videoStyle').value : 'cinematic';
  var fullPrompt = prompt + ', ' + styleVal + ' film still, cinematic lighting, 4k, detailed';
  var seed       = Math.floor(Math.random() * 999999);
  var imgUrl     = 'https://image.pollinations.ai/prompt/' + encodeURIComponent(fullPrompt) + '?width=1280&height=720&seed=' + seed + '&nologo=true';

  var previewImg    = document.createElement('img');
  previewImg.id     = 'vidResult';
  previewImg.style.cssText = 'display:none;position:absolute;inset:0;width:100%;height:100%;object-fit:contain;z-index:10;border-radius:10px;';
  vb.appendChild(previewImg);

  previewImg.onload = function() {
    previewImg.onload = null;
    previewImg.onerror = null;
    vlc.style.display  = 'none';
    previewImg.style.display = 'block';
    addAsset({ type: 'image', src: imgUrl, name: prompt.substring(0, 30) });
    showToast('Frame generated!', 'ok');
  };

  previewImg.onerror = function() {
    previewImg.onload = null;
    previewImg.onerror = null;
    var retryUrl = 'https://image.pollinations.ai/prompt/' + encodeURIComponent(fullPrompt) + '?width=1280&height=720&seed=' + (seed + 1) + '&nologo=true';
    previewImg.onload = function() {
      previewImg.onload = null;
      previewImg.onerror = null;
      vlc.style.display = 'none';
      previewImg.style.display = 'block';
      addAsset({ type: 'image', src: retryUrl, name: prompt.substring(0, 30) });
      showToast('Frame generated!', 'ok');
    };
    previewImg.onerror = function() {
      previewImg.onload = null;
      previewImg.onerror = null;
      vlc.style.display = 'none';
      ves.style.display = '';
      showToast('Generation failed - check connection', 'err');
    };
    previewImg.src = retryUrl;
  };

  previewImg.src = imgUrl;
}

// Voiceover functions
function generateVoiceover() {
  var script = document.getElementById('voScript').value.trim();
  var voice  = document.getElementById('voiceSelect').value;
  var apiKey = localStorage.getItem('elevenLabsApiKey');

  if (!script) { showToast('Please enter a script', 'err'); updateStatus('error', 'No script provided'); return; }
  if (!voice)  { showToast('Please select a voice', 'err'); updateStatus('error', 'No voice selected'); return; }
  if (!apiKey) { showToast('Please save your API key first', 'err'); updateStatus('error', 'No API key found'); return; }

  updateStatus('processing', 'Generating voiceover...');

  // ElevenLabs voice ID map
  var customId = localStorage.getItem('customVoiceId') || '';
  var voiceIds = {
    rachel:  '21m00Tcm4TlvDq8ikWAM',
    domi:    'AZnzlk1XvdvUeBnXmlld',
    bella:   'EXAVITQu4vr4xnSDxMaL',
    antoni:  'ErXwobaYiN019PkySvjV',
    arnold:  'VR6AewLTigWG4xSOukaG',
    adam:    'pNInz6obpgDQGcFmaJgB',
    sam:     'yoZ06aMxZJJ28mfd3POQ',
    custom:  customId
  };

  if (voice === 'custom' && !customId) {
    showToast('Open Settings and enter your custom voice ID', 'err');
    updateStatus('error', 'No custom voice ID set');
    return;
  }

  var vid    = voiceIds[voice] || voiceIds['rachel'];
  var apiUrl = 'https://api.elevenlabs.io/v1/text-to-speech/' + vid;

  var stability = 0.5;
  var clarity   = 0.75;
  var stabEl = document.getElementById('voiceStability');
  var clarEl = document.getElementById('voiceClarity');
  if (stabEl) stability = parseFloat(stabEl.value) / 100;
  if (clarEl) clarity   = parseFloat(clarEl.value) / 100;

  fetch(apiUrl, {
    method: 'POST',
    headers: {
      'xi-api-key': apiKey,
      'Content-Type': 'application/json',
      'Accept': 'audio/mpeg'
    },
    body: JSON.stringify({
      text: script,
      model_id: 'eleven_monolingual_v1',
      voice_settings: { stability: stability, similarity_boost: clarity }
    })
  })
  .then(function(r) {
    if (!r.ok) { throw new Error('API error ' + r.status); }
    return r.blob();
  })
  .then(function(blob) {
    var url    = URL.createObjectURL(blob);
    var audio  = document.getElementById('aPrev');
    var audBox = document.getElementById('audBox');
    audio.pause();
    audio.src = url;
    audio.load();
    audBox.classList.add('on');
    updateStatus('ready', 'Ready - press play');
    showToast('Voiceover ready!', 'ok');
  })
  .catch(function(err) {
    updateStatus('error', 'API error: ' + err.message);
    showToast('API error - check your key', 'err');
  });
}

// Nyxen send message
function sendNyxen() {
  var inp = document.getElementById('nyxenTextIn');
  var msg = inp.value.trim();
  if (!msg) return;
  appendNyxenMsg(msg, 'user');
  inp.value = '';
  fetchNyxen(msg);
}

function appendNyxenMsg(text, role) {
  var msgs = document.getElementById('nyxenMsgs');
  var d = document.createElement('div');
  d.className = 'nyxen-msg ' + role;
  if (role === 'ai') {
    d.innerHTML = '<div class="nlbl">NYXEN</div>' + text;
  } else {
    d.textContent = text;
  }
  msgs.appendChild(d);
  msgs.scrollTop = msgs.scrollHeight;
}

function fetchNyxen(msg) {
  // Typing indicator
  var msgs = document.getElementById('nyxenMsgs');
  var typing = document.createElement('div');
  typing.className = 'nyxen-msg ai';
  typing.id = 'nyxenTyping';
  typing.innerHTML = '<div class="nlbl">NYXEN</div><span style="opacity:.5;font-style:italic;">typing...</span>';
  msgs.appendChild(typing);
  msgs.scrollTop = msgs.scrollHeight;

  fetch('https://nyxen.smcantrellbooks.workers.dev/nyxen', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ message: msg })
  })
  .then(function(r) { return r.json(); })
  .then(function(d) {
    var t = document.getElementById('nyxenTyping');
    if (t) t.remove();
    var reply = d.reply || d.message || d.response || d.text || 'I\'m here! How can I help?';
    appendNyxenMsg(reply, 'ai');
    speakNyxen(reply);
  })
  .catch(function() {
    var t = document.getElementById('nyxenTyping');
    if (t) t.remove();
    appendNyxenMsg('I\'m here! Let me know how I can help.', 'ai');
  });
}

// Nyxen voice output — via Cloudflare Worker (key stored server-side)
function speakNyxen(text) {
  fetch('https://nyxen.smcantrellbooks.workers.dev/nyxen', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ message: text, tts: true })
  })
  .then(function(r) {
    // If worker returns audio blob
    var ct = r.headers.get('content-type') || '';
    if (ct.includes('audio')) {
      return r.blob().then(function(blob) {
        var url = URL.createObjectURL(blob);
        var audio = new Audio(url);
        audio.play();
        audio.onended = function() { URL.revokeObjectURL(url); };
      });
    }
    // Worker returned JSON — no audio in response, silent
    return r.json().catch(function(){});
  })
  .catch(function() {
    // Silent fail — chat still works
  });
}

// Nyxen voice input (Web Speech Recognition)
var nyxenRecognition = null;
function toggleNyxenVoice() {
  var btn = document.getElementById('nyxenMicBtn');
  if (!('webkitSpeechRecognition' in window || 'SpeechRecognition' in window)) {
    showToast('Voice input not supported in this browser', 'err');
    return;
  }
  if (nyxenRecognition && btn.classList.contains('listening')) {
    nyxenRecognition.stop();
    btn.classList.remove('listening');
    btn.textContent = '🎤';
    return;
  }
  var SR = window.SpeechRecognition || window.webkitSpeechRecognition;
  nyxenRecognition = new SR();
  nyxenRecognition.lang = 'en-US';
  nyxenRecognition.interimResults = false;
  nyxenRecognition.maxAlternatives = 1;

  nyxenRecognition.onstart = function() {
    btn.classList.add('listening');
    btn.textContent = '⏹';
  };
  nyxenRecognition.onresult = function(e) {
    var transcript = e.results[0][0].transcript;
    document.getElementById('nyxenTextIn').value = transcript;
    btn.classList.remove('listening');
    btn.textContent = '🎤';
    sendNyxen();
  };
  nyxenRecognition.onerror = function() {
    btn.classList.remove('listening');
    btn.textContent = '🎤';
    showToast('Voice input error — try again', 'err');
  };
  nyxenRecognition.onend = function() {
    btn.classList.remove('listening');
    btn.textContent = '🎤';
  };
  nyxenRecognition.start();
}
