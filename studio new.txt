var imgTimeout = null;

function generateImage(){
  var prompt = document.getElementById('promptInput').value.trim();
  if(!prompt){ showToast('Please enter a description','err'); return; }
  var lc=document.getElementById('LC'), es=document.getElementById('ES'), img=document.getElementById('cImg');
  var bgVid = document.getElementById('bgVideo');
  es.classList.add('hide');
  img.style.display='none'; img.classList.remove('vis');
  lc.classList.add('show');

  var style = document.getElementById('imageStyle').value;
  var fullPrompt = prompt+', '+style+', high quality, detailed, professional';
  var seed = Math.floor(Math.random()*999999);
  var ratio = document.getElementById('imageRatio').value;
  var w=1280,h=720;
  if(ratio==='2/3'){w=800;h=1200;}else if(ratio==='1/1'){w=1024;h=1024;}else if(ratio==='4/3'){w=1024;h=768;}else if(ratio==='21/9'){w=1680;h=720;}

  // Use multiple image services as fallback
  var services = [
    'https://image.pollinations.ai/prompt/'+encodeURIComponent(fullPrompt)+'?width='+w+'&height='+h+'&seed='+seed+'&nologo=true&enhance=true',
    'https://image.pollinations.ai/prompt/'+encodeURIComponent(fullPrompt)+'?width='+w+'&height='+h+'&seed='+(seed+1)+'&nologo=true',
    'https://image.pollinations.ai/prompt/'+encodeURIComponent(prompt+', '+style)+'?width='+w+'&height='+h+'&seed='+(seed+2)+'&nologo=true'
  ];
  var attempt = 0;

  function tryLoad(){
    if(attempt >= services.length){
      lc.classList.remove('show'); es.classList.remove('hide');
      showToast('Generation failed — please try again','err');
      return;
    }
    var url = services[attempt++];
    var loader = new Image();
    loader.onload = function(){
      if(imgTimeout) clearTimeout(imgTimeout);
      lc.classList.remove('show');
      // Hide background video
      if(bgVid) bgVid.style.display='none';
      img.src=url;
      img.style.cssText='display:block!important;position:absolute;inset:0;width:100%;height:100%;object-fit:contain;z-index:5;border-radius:12px;';
      img.classList.add('vis');
      es.classList.add('hide');
      addAsset({type:'image',src:url,name:prompt.substring(0,30)});
      addRecentImage(url,prompt.substring(0,20));
      applyAdjustments();
      showToast('Image generated!','ok');
    };
    loader.onerror = function(){ tryLoad(); };
    loader.src = url;
  }

  if(imgTimeout) clearTimeout(imgTimeout);
  imgTimeout = setTimeout(function(){
    if(lc.classList.contains('show')){
      lc.classList.remove('show'); es.classList.remove('hide');
      showToast('Taking too long — try a simpler prompt','err');
    }
  }, 45000);

  tryLoad();
}

function applyAdjustments(){
  var img = document.getElementById('cImg');
  if(!img || !img.classList.contains('vis')) return;
  var brightness = document.getElementById('cebBrightness') ? document.getElementById('cebBrightness').value : document.getElementById('brightnessSlider').value;
  var contrast = document.getElementById('cebContrast') ? document.getElementById('cebContrast').value : document.getElementById('contrastSlider').value;
  var saturation = document.getElementById('cebSaturation') ? document.getElementById('cebSaturation').value : document.getElementById('saturationSlider').value;
  var opacity = document.getElementById('cebOpacity') ? document.getElementById('cebOpacity').value : document.getElementById('opacitySlider').value;

  var filters = 'brightness('+brightness+'%) contrast('+contrast+'%) saturate('+saturation+'%)';
  switch(currentFilter){
    case 'cinematic': filters += ' contrast(120%) brightness(90%) saturate(80%)'; break;
    case 'noir': filters += ' grayscale(100%) contrast(130%)'; break;
    case 'warm': filters += ' sepia(30%) saturate(120%)'; break;
    case 'cool': filters += ' hue-rotate(180deg) saturate(80%)'; break;
    case 'vintage': filters += ' sepia(50%) contrast(90%) brightness(90%)'; break;
    case 'dramatic': filters += ' contrast(150%) brightness(85%) saturate(130%)'; break;
    case 'fade': filters += ' brightness(120%) contrast(80%) saturate(70%)'; break;
  }
  var transform = 'rotate('+currentRotation+'deg) scaleX('+(flipH?-1:1)+') scaleY('+(flipV?-1:1)+')';
  img.style.filter = filters;
  img.style.transform = transform;
  img.style.opacity = opacity/100;
}

function rotateImg(deg){
  currentRotation = (currentRotation + deg) % 360;
  applyAdjustments();
}

function flipImg(axis){
  if(axis==='h') flipH = !flipH;
  else flipV = !flipV;
  applyAdjustments();
}

function syncSlider(type, val){
  // Keep sidebar sliders in sync with canvas bar sliders
  var sidebarMap = {brightness:'brightnessSlider',contrast:'contrastSlider',saturation:'saturationSlider',opacity:'opacitySlider'};
  var valMap = {brightness:'brightnessVal',contrast:'contrastVal',saturation:'saturationVal',opacity:'opacityVal'};
  var barMap = {brightness:'cebBrightness',contrast:'cebContrast',saturation:'cebSaturation',opacity:'cebOpacity'};
  var barValMap = {brightness:'cebBrightnessVal',contrast:'cebContrastVal',saturation:'cebSaturationVal',opacity:'cebOpacityVal'};
  var suffix = type==='opacity'?'%':'%';
  var sb = document.getElementById(sidebarMap[type]); if(sb) sb.value=val;
  var sbv = document.getElementById(valMap[type]); if(sbv) sbv.textContent=val+suffix;
  var bb = document.getElementById(barMap[type]); if(bb) bb.value=val;
  var bbv = document.getElementById(barValMap[type]); if(bbv) bbv.textContent=val;
}

var dragEl=null, dragOffX=0, dragOffY=0;
function startDrag(e){
  if(e.target.tagName==='BUTTON'||e.target.tagName==='INPUT'||e.target.tagName==='SELECT') return;
  dragEl=document.getElementById('floatEditPanel');
  dragOffX=e.clientX-dragEl.getBoundingClientRect().left;
  dragOffY=e.clientY-dragEl.getBoundingClientRect().top;
  document.onmousemove=doDrag; document.onmouseup=stopDrag;
}
function doDrag(e){
  if(!dragEl) return;
  dragEl.style.left=Math.max(0,e.clientX-dragOffX)+'px';
  dragEl.style.top=Math.max(0,e.clientY-dragOffY)+'px';
  dragEl.style.right='auto';
}
function stopDrag(){ dragEl=null; document.onmousemove=null; document.onmouseup=null; }

/* WAVEFORM VISUALIZER */
var waveAnimId = null;
var waveStart = Date.now();

function initWaveform(){
  var canvas = document.getElementById('waveCanvas');
  if(!canvas) return;
  var ctx = canvas.getContext('2d');
  var bars = 80;

  function draw(){
    canvas.width = canvas.offsetWidth;
    canvas.height = canvas.offsetHeight;
    var w = canvas.width, h = canvas.height;
    ctx.clearRect(0,0,w,h);
    var t = Date.now() / 1000;
    var barW = w / bars;

    for(var i=0; i<bars; i++){
      var x = i * barW;
      // Multi-frequency wave with gold/plum gradient
      var amp = (
        Math.sin(t * 2.1 + i * 0.18) * 0.35 +
        Math.sin(t * 3.7 + i * 0.31) * 0.25 +
        Math.sin(t * 1.4 + i * 0.08) * 0.20 +
        Math.sin(t * 5.2 + i * 0.44) * 0.10 +
        Math.random() * 0.10
      );
      var barH = Math.max(3, Math.abs(amp) * h * 0.85);
      var y = (h - barH) / 2;

      // Gold to plum gradient per bar
      var grad = ctx.createLinearGradient(0, y, 0, y+barH);
      var intensity = Math.abs(amp);
      if(intensity > 0.4){
        grad.addColorStop(0, 'rgba(201,160,64,0.9)');
        grad.addColorStop(0.5, 'rgba(232,192,96,1)');
        grad.addColorStop(1, 'rgba(201,160,64,0.9)');
      } else {
        grad.addColorStop(0, 'rgba(138,70,99,0.5)');
        grad.addColorStop(0.5, 'rgba(176,88,120,0.7)');
        grad.addColorStop(1, 'rgba(138,70,99,0.5)');
      }
      ctx.fillStyle = grad;
      ctx.beginPath();
      ctx.roundRect(x+1, y, Math.max(1, barW-2), barH, 2);
      ctx.fill();
    }

    // Timer
    var elapsed = Math.floor((Date.now() - waveStart) / 1000);
    var m = Math.floor(elapsed/60).toString().padStart(2,'0');
    var s = (elapsed%60).toString().padStart(2,'0');
    var timerEl = document.getElementById('waveTimer');
    if(timerEl) timerEl.textContent = m+':'+s;

    waveAnimId = requestAnimationFrame(draw);
  }
  draw();
}

window.addEventListener('DOMContentLoaded',function(){
  setTimeout(initWaveform, 500);
  var savedKey=localStorage.getItem('elevenLabsApiKey');
  if(savedKey) document.getElementById('apiKey').value=savedKey;
  var savedVoiceId=localStorage.getItem('customVoiceId');
  if(savedVoiceId) document.getElementById('customVoiceId').value=savedVoiceId;
});
